class Plant
  parameter Real tau = 0.05 "Time constant (seconds)";
  Real x(start = 0.0) "Plant state variable";
  input Real u "Control input";
  parameter Real stepVoltage.V(quantity = "ElectricPotential", unit = "V", start = 1.0) = 5.0 "Height of step";
  Real stepVoltage.v(quantity = "ElectricPotential", unit = "V") "Voltage drop of the two pins (= p.v - n.v)";
  Real stepVoltage.p.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real stepVoltage.p.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
  Real stepVoltage.n.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real stepVoltage.n.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
  Real stepVoltage.i(quantity = "ElectricCurrent", unit = "A") "Current flowing from pin p to pin n";
  final parameter Real stepVoltage.signalSource.height = stepVoltage.V "Height of step";
  Real stepVoltage.signalSource.y "Connector of Real output signal";
  final parameter Real stepVoltage.signalSource.offset = stepVoltage.offset "Offset of output signal y";
  final parameter Real stepVoltage.signalSource.startTime(quantity = "Time", unit = "s") = stepVoltage.startTime "Output y = offset for time < startTime";
  parameter Real stepVoltage.offset(quantity = "ElectricPotential", unit = "V") = 0.0 "Voltage offset";
  parameter Real stepVoltage.startTime(quantity = "Time", unit = "s") = 0.0 "Time offset";
  parameter Real resistor1.R(quantity = "Resistance", unit = "Ohm", start = 1.0) = 10.0 "Resistance at temperature T_ref";
  parameter Real resistor1.T_ref(quantity = "ThermodynamicTemperature", unit = "K", displayUnit = "degC", min = 0.0, start = 288.15, nominal = 300.0) = 300.15 "Reference temperature";
  parameter Real resistor1.alpha(quantity = "LinearTemperatureCoefficient", unit = "1/K") = 0.0 "Temperature coefficient of resistance (R_actual = R*(1 + alpha*(T_heatPort - T_ref))";
  Real resistor1.v(quantity = "ElectricPotential", unit = "V") "Voltage drop of the two pins (= p.v - n.v)";
  Real resistor1.p.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real resistor1.p.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
  Real resistor1.n.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real resistor1.n.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
  Real resistor1.i(quantity = "ElectricCurrent", unit = "A") "Current flowing from pin p to pin n";
  final parameter Boolean resistor1.useHeatPort = false "= true, if heatPort is enabled";
  parameter Real resistor1.T(quantity = "ThermodynamicTemperature", unit = "K", displayUnit = "degC", min = 0.0, start = 288.15, nominal = 300.0) = resistor1.T_ref "Fixed device temperature if useHeatPort = false";
  Real resistor1.LossPower(quantity = "Power", unit = "W") "Loss power leaving component via heatPort";
  Real resistor1.T_heatPort(quantity = "ThermodynamicTemperature", unit = "K", displayUnit = "degC", min = 0.0, start = 288.15, nominal = 300.0) "Temperature of heatPort";
  Real resistor1.R_actual(quantity = "Resistance", unit = "Ohm") "Actual resistance = R*(1 + alpha*(T_heatPort - T_ref))";
  Real capacitor1.v(quantity = "ElectricPotential", unit = "V", start = 0.0) "Voltage drop of the two pins (= p.v - n.v)";
  Real capacitor1.p.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real capacitor1.p.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
  Real capacitor1.n.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real capacitor1.n.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
  Real capacitor1.i(quantity = "ElectricCurrent", unit = "A") "Current flowing from pin p to pin n";
  parameter Real capacitor1.C(quantity = "Capacitance", unit = "F", min = 0.0, start = 1.0) = 0.01 "Capacitance";
  Real ground1.p.v(quantity = "ElectricPotential", unit = "V") "Potential at the pin";
  Real ground1.p.i(quantity = "ElectricCurrent", unit = "A") "Current flowing into the pin";
equation
  stepVoltage.n.v = capacitor1.p.v;
  stepVoltage.p.v = resistor1.p.v;
  capacitor1.n.v = ground1.p.v;
  capacitor1.n.v = resistor1.n.v;
  resistor1.p.i + stepVoltage.p.i = 0.0;
  capacitor1.p.i + stepVoltage.n.i = 0.0;
  ground1.p.i + capacitor1.n.i + resistor1.n.i = 0.0;
  stepVoltage.signalSource.y = stepVoltage.signalSource.offset + (if time < stepVoltage.signalSource.startTime then 0.0 else stepVoltage.signalSource.height);
  stepVoltage.v = stepVoltage.signalSource.y;
  0.0 = stepVoltage.p.i + stepVoltage.n.i;
  stepVoltage.i = stepVoltage.p.i;
  stepVoltage.v = stepVoltage.p.v - stepVoltage.n.v;
  assert(1.0 + resistor1.alpha * (resistor1.T_heatPort - resistor1.T_ref) >= 2.220446049250313e-16, "Temperature outside scope of model!");
  resistor1.R_actual = resistor1.R * (1.0 + resistor1.alpha * (resistor1.T_heatPort - resistor1.T_ref));
  resistor1.v = resistor1.R_actual * resistor1.i;
  resistor1.LossPower = resistor1.v * resistor1.i;
  resistor1.T_heatPort = resistor1.T;
  0.0 = resistor1.p.i + resistor1.n.i;
  resistor1.i = resistor1.p.i;
  resistor1.v = resistor1.p.v - resistor1.n.v;
  capacitor1.i = capacitor1.C * der(capacitor1.v);
  0.0 = capacitor1.p.i + capacitor1.n.i;
  capacitor1.i = capacitor1.p.i;
  capacitor1.v = capacitor1.p.v - capacitor1.n.v;
  ground1.p.v = 0.0;
  der(x) = (u - x) / tau;
end Plant;
